#ScenePlay
server {
	listen 80 default_server;
	listen [::]:80 default_server;

	server_name _;

	# No upload size cap (nginx defaults to 1m -> "413 Request Entity Too
	# Large"): battlemap videos, AI-generated art, and backup-restore zips
	# all exceed 1m routinely. LAN-only app, so unlimited is fine here.
	client_max_body_size 0;

	# Live updates (Server-Sent Events): no buffering, long read timeout, HTTP/1.1
	# without the Connection header so the stream stays open through the proxy.
	location /api/v1/events {
             proxy_pass http://127.0.0.1:8086;
             proxy_http_version 1.1;
             proxy_set_header Connection '';
             proxy_buffering off;
             proxy_cache off;
             proxy_read_timeout 3600s;
             proxy_set_header Host $host;
             proxy_set_header X-Real-IP $remote_addr;
             proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
             proxy_set_header X-Forwarded-Proto $scheme;
        }

	location / {
             proxy_pass http://127.0.0.1:8086;
             proxy_set_header Host $host;
             proxy_set_header X-Real-IP $remote_addr;
             proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
             proxy_set_header X-Forwarded-Proto $scheme;
        }
}
